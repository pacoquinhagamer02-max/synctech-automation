🔄 Sincronizar entre celular e PC

Sem isso, os leads que você cadastra ficam só no aparelho usado. Com 5 min de setup (planilha Google), tudo aparece em qualquer aparelho.
Ver passo a passo →
1
Crie uma planilha Google nova
sheets.google.com → planilha em branco. Pode deixar sem nenhuma aba especial, o script cria sozinho.
2
Cole o script em Extensões → Apps Script
Apague o conteúdo padrão e cole o código abaixo. Depois clique em Implantar → Nova implantação → tipo "App da Web", com acesso "Qualquer pessoa" — copie a URL gerada.
3
Cole a URL aqui embaixo em todo aparelho
Repita esse último passo (só colar a URL, o script é o mesmo) em cada celular/PC onde for usar o Pipeline.
function doPost(e) { var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Pipeline') || SpreadsheetApp.getActiveSpreadsheet().insertSheet('Pipeline'); if (sh.getLastRow() === 0) sh.appendRow(['id','name','biz','phone','city','src','val','stage','time']); var data = JSON.parse(e.postData.contents); if (data.action === 'add') { var l = data.lead; sh.appendRow([l.id, l.name, l.biz, l.phone, l.city, l.src, l.val, l.stage, l.time]); } else if (data.action === 'update') { var rows = sh.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] === data.id) { sh.getRange(i+1, 8).setValue(data.stage); sh.getRange(i+1, 9).setValue(data.time || ''); break; } } } return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Pipeline'); if (!sh || sh.getLastRow() < 2) return ContentService.createTextOutput('[]').setMimeType(ContentService.MimeType.JSON); var rows = sh.getDataRange().getValues(); var headers = rows[0]; var leads = rows.slice(1).map(function(row) { var obj = {}; headers.forEach(function(h, i) { obj[h] = row[i]; }); return obj; }); return ContentService.createTextOutput(JSON.stringify(leads)).setMimeType(ContentService.MimeType.JSON); }
